В-5 Витрина «Фастфуд и Общепит»: служебный путь для карточек

Страница «1C:Предприятие 8. Фастфуд и Общепит» · https://alsn.ru/1_obschepit_fastfood · адрес в Тильде 1_obschepit_fastfood

Что сделать

Вставить в HEAD витрины тот же универсальный скрипт, что уже работает на «Лицензиях 1С». Он сам собирает путь для каждой карточки товара этой витрины.

Зачем

На карточках товаров этой витрины поисковик сейчас не видит путь «Главная / раздел / товар». На других витринах он уже есть. Скрипт один на витрину, новые товары подхватываются сами.

Как в Тильде
  1. Список страниц → страница → Настройки (шестерёнка) → Дополнительно → поле «HTML-код для вставки внутрь head» (1_obschepit_fastfood). Это настройки страницы витрины, не HEAD сайта.
  2. Встать в конец поля, ничего не стирая, вставить:
<script>
(function () {
  if (!/\/tproduct\//.test(location.pathname || "")) return;
  if (document.getElementById("alsn-bc-jsonld")) return;

  function txt(el) {
    return (el && (el.textContent || "")).replace(/\s+/g, " ").trim();
  }
  function abs(href) {
    try { return new URL(href, location.origin).href.split("?")[0]; } catch (e) { return href; }
  }
  function pageUrl() {
    var c = document.querySelector('link[rel="canonical"]');
    var u = (c && c.href) || (location.origin + location.pathname);
    return u.split("?")[0];
  }
  function productName() {
    return txt(document.querySelector("h1")) ||
      txt(document.querySelector('[itemprop="name"]')) ||
      txt(document.querySelector(".js-store-prod-name, .t-store__prod-popup__name")) ||
      (document.title || "").split("|")[0].trim();
  }
  function catalogUrl() {
    var p = (location.pathname || "").replace(/\/tproduct\/.*$/, "") || "/";
    if (p.length > 1 && p.charAt(p.length - 1) === "/") p = p.slice(0, -1);
    return location.origin + p;
  }
  function visCrumbs() {
    var root = document.querySelector(".t-catalog__breadcrumbs");
    if (!root) return [];
    var nodes = root.querySelectorAll("a, span");
    var out = [];
    for (var i = 0; i < nodes.length; i++) {
      var el = nodes[i];
      var name = txt(el);
      if (!name) continue;
      var href = el.tagName === "A" ? el.getAttribute("href") : "";
      out.push({ name: name, item: href ? abs(href) : "" });
    }
    return out;
  }
  function build() {
    if (document.getElementById("alsn-bc-jsonld")) return true;
    var name = productName();
    if (!name) return false;
    var vis = visCrumbs();
    var list = [{ name: "Главная", item: location.origin + "/" }];
    var seen = { "Главная": true };
    var i, row;
    for (i = 0; i < vis.length; i++) {
      row = vis[i];
      if (!row.name || seen[row.name]) continue;
      seen[row.name] = true;
      list.push(row);
    }
    if (list.length === 1) {
      list.push({ name: "Каталог", item: catalogUrl() });
    }
    if (list[1] && !list[1].item) list[1].item = catalogUrl();
    if (list[list.length - 1].name !== name) {
      list.push({ name: name, item: pageUrl() });
    } else {
      list[list.length - 1].item = pageUrl();
    }
    var els = [];
    for (i = 0; i < list.length; i++) {
      els.push({ "@type": "ListItem", position: i + 1, name: list[i].name, item: list[i].item });
    }
    var s = document.createElement("script");
    s.type = "application/ld+json";
    s.id = "alsn-bc-jsonld";
    s.text = JSON.stringify({
      "@context": "https://schema.org",
      "@type": "BreadcrumbList",
      "@id": pageUrl() + "#breadcrumb",
      itemListElement: els
    });
    document.head.appendChild(s);
    return true;
  }
  function start() {
    var n = 0;
    function tick() {
      n++;
      var hasNav = !!document.querySelector(".t-catalog__breadcrumbs");
      var hasName = !!productName();
      if ((hasNav && hasName) || n > 40) {
        build();
        return;
      }
      setTimeout(tick, 250);
    }
    tick();
  }
  if (document.readyState === "loading") document.addEventListener("DOMContentLoaded", start);
  else start();
})();
</script>
  1. Сохранить.
  2. Видимые крошки на карточках дают блоки каталога ST340: в блоке каталога на витрине → Контент → вкладка «Хлебные крошки» → «Над заголовком». Если уже включено, не трогать.

В-7 Поставщики B2B: заменить старые крошки

Что сделать

Убрать старую строку «Интеграция 1C по API с B2B поставщиками и дистрибьютерами → …» и поставить крошки «дом / Интеграция с поставщиками / Имя поставщика», как уже сделано на странице Мерлион. Добавить служебный путь в HEAD.

Зачем

Это страницы продукта «Интеграция с B2B-поставщиками телеком-оборудования». Новые крошки ведут гостя на общую страницу продукта, а поисковик видит, что все партнёры - часть одного раздела. Эталон на сайте: https://alsn.ru/merlion.

Где

Сразу под меню стоит старый блок со стрелкой, под ним баннер T180 «Однократная оплата без ежегодного продления! Интеграция 1C с …». Верх баннера светлый, поэтому крошки серые на белом.

Как в Тильде (на каждой странице)

  2. Если под меню есть блок со стрелкой → (T758) - навести → корзина / «Удалить».
  3. Навести на первый экран с заголовком → «+» сверху → Библиотека блоков → Другое → T123 «HTML-код».
  4. Контент T123 → вставить код «На экран» этой страницы.
  5. Настройки T123 → цвет фона из карточки страницы, отступы сверху и снизу 0. Сохранить.
  6. Список страниц → страница → Настройки (шестерёнка) → Дополнительно → поле «HTML-код для вставки внутрь head» → в конец, ничего не стирая, вставить код «В HEAD». Сохранить.

В-8 Переходы и синхронизации: заменить старые крошки

Что сделать

Убрать старую строку «Сопровождение 1С → …» и поставить крошки «дом / раздел / короткое имя». Переходы на новую конфигурацию - в разделе «Внедрение 1С», как уже сделано на странице перехода с УПП. Синхронизации баз - в разделе «Техподдержка 1С». Добавить служебный путь в HEAD.

Зачем

Старая строка ведёт на «Внедрение 1С» с подписью «Сопровождение 1С», это путает гостя. Новый путь честно показывает раздел, а поисковик связывает страницу с нужной услугой. Эталон: https://alsn.ru/perehod-s-upp-na-ka-unf-ut.

Где

На шести страницах под меню старый блок со стрелкой, под ним баннер T180 на белом фоне: крошки серые, фон белый. На «Переход с МойСклад» старого блока нет, первый экран - фото с тёмной вуалью: крошки светло-серые, фон T123 чёрный. Если раздел для синхронизаций хотите другой (например «Внедрение 1С»), скажите до вставки.

Как в Тильде (на каждой странице)
  1. Список страниц → адрес страницы → Редактировать.
  2. Если под меню есть блок со стрелкой → (T758) - навести → корзина / «Удалить».
  3. Навести на первый экран с заголовком → «+» сверху → Библиотека блоков → Другое → T123 «HTML-код».
  4. Контент T123 → вставить код «На экран» этой страницы.
  5. Настройки T123 → цвет фона из карточки страницы, отступы сверху и снизу 0. Сохранить.
  6. Список страниц → страница → Настройки (шестерёнка) → Дополнительно → поле «HTML-код для вставки внутрь head» → в конец, ничего не стирая, вставить код «В HEAD». Сохранить.
