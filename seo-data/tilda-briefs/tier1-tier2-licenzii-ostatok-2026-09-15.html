D Служебный путь для карточек Каталога (образец)

Что сделано

В HEAD страницы витрины «Лицензии 1С» (https://alsn.ru/dopolnitelnie_licenzii) стоит один универсальный скрипт. На каждой карточке товара (/tproduct/) он сам собирает служебную навигацию: «Главная», звенья из видимых крошек ST340 и имя товара из заголовка. На самой витрине скрипт ничего не делает. Новые товары покрываются без правок.

Зачем

У карточки Каталога нет своего HEAD, а Тильда копирует HEAD витрины на все её карточки. Поэтому один скрипт вместо списка товаров: Яндекс и Google видят путь «Главная / Лицензии 1С / товар», как гость видит его над заголовком.

Как в Тильде (для новой витрины)
  1. Открыть страницу витрины → шестерёнка «Настройки страницы» → Дополнительно (или «Вставка кода»).
  2. Поле «HTML-код для вставки внутрь HEAD» → курсор в самый конец, ничего не стирать.
  3. Вставить код ниже как есть → Сохранить → Опубликовать страницу витрины.
  4. У карточек включить крошки: блок ST340 → вкладка «Хлебные крошки» → «Над заголовком».
<script>
(function () {
  if (!/\/tproduct\//.test(location.pathname || "")) return;
  if (document.getElementById("alsn-bc-jsonld")) return;

  function txt(el) {
    return (el && (el.textContent || "")).replace(/\s+/g, " ").trim();
  }
  function abs(href) {
    try { return new URL(href, location.origin).href.split("?")[0]; } catch (e) { return href; }
  }
  function pageUrl() {
    var c = document.querySelector('link[rel="canonical"]');
    var u = (c && c.href) || (location.origin + location.pathname);
    return u.split("?")[0];
  }
  function productName() {
    return txt(document.querySelector("h1")) ||
      txt(document.querySelector('[itemprop="name"]')) ||
      txt(document.querySelector(".js-store-prod-name, .t-store__prod-popup__name")) ||
      (document.title || "").split("|")[0].trim();
  }
  function catalogUrl() {
    var p = (location.pathname || "").replace(/\/tproduct\/.*$/, "") || "/";
    if (p.length > 1 && p.charAt(p.length - 1) === "/") p = p.slice(0, -1);
    return location.origin + p;
  }
  function visCrumbs() {
    var root = document.querySelector(".t-catalog__breadcrumbs");
    if (!root) return [];
    var nodes = root.querySelectorAll("a, span");
    var out = [];
    for (var i = 0; i < nodes.length; i++) {
      var el = nodes[i];
      var name = txt(el);
      if (!name) continue;
      var href = el.tagName === "A" ? el.getAttribute("href") : "";
      out.push({ name: name, item: href ? abs(href) : "" });
    }
    return out;
  }
  function build() {
    if (document.getElementById("alsn-bc-jsonld")) return true;
    var name = productName();
    if (!name) return false;
    var vis = visCrumbs();
    var list = [{ name: "Главная", item: location.origin + "/" }];
    var seen = { "Главная": true };
    var i, row;
    for (i = 0; i < vis.length; i++) {
      row = vis[i];
      if (!row.name || seen[row.name]) continue;
      seen[row.name] = true;
      list.push(row);
    }
    if (list.length === 1) {
      list.push({ name: "Каталог", item: catalogUrl() });
    }
    if (list[1] && !list[1].item) list[1].item = catalogUrl();
    if (list[list.length - 1].name !== name) {
      list.push({ name: name, item: pageUrl() });
    } else {
      list[list.length - 1].item = pageUrl();
    }
    var els = [];
    for (i = 0; i < list.length; i++) {
      els.push({ "@type": "ListItem", position: i + 1, name: list[i].name, item: list[i].item });
    }
    var s = document.createElement("script");
    s.type = "application/ld+json";
    s.id = "alsn-bc-jsonld";
    s.text = JSON.stringify({
      "@context": "https://schema.org",
      "@type": "BreadcrumbList",
      "@id": pageUrl() + "#breadcrumb",
      itemListElement: els
    });
    document.head.appendChild(s);
    return true;
  }
  function start() {
    var n = 0;
    function tick() {
      n++;
      var hasNav = !!document.querySelector(".t-catalog__breadcrumbs");
      var hasName = !!productName();
      if ((hasNav && hasName) || n > 40) {
        build();
        return;
      }
      setTimeout(tick, 250);
    }
    tick();
  }
  if (document.readyState === "loading") document.addEventListener("DOMContentLoaded", start);
  else start();
})();
</script>
